-- claim: one manifest row per candidate URL, no PDF content (sha256) downloaded twice, every 'duplicate' points
--        to a downloaded URL, downloaded rows have a real file
-- returns: offending manifest rows
SELECT 'duplicate_source_url' AS violation, source_url AS key, count(*) AS n
FROM classic_stable.auto_intelligence.market_download_manifest GROUP BY source_url HAVING count(*) > 1
UNION ALL
SELECT 'same_content_downloaded_twice', sha256, count(*)
FROM classic_stable.auto_intelligence.market_download_manifest WHERE status = 'downloaded' AND sha256 IS NOT NULL
GROUP BY sha256 HAVING count(*) > 1
UNION ALL
SELECT 'downloaded_without_file', source_url, 1
FROM classic_stable.auto_intelligence.market_download_manifest
WHERE status = 'downloaded' AND (file_path IS NULL OR bytes IS NULL OR bytes < 10000 OR sha256 IS NULL)
UNION ALL
SELECT 'duplicate_without_original', d.source_url, 1
FROM classic_stable.auto_intelligence.market_download_manifest d
LEFT JOIN classic_stable.auto_intelligence.market_download_manifest o ON o.source_url = d.duplicate_of AND o.status = 'downloaded'
WHERE d.status = 'duplicate' AND (o.source_url IS NULL OR d.sha256 IS DISTINCT FROM o.sha256)
UNION ALL
SELECT 'unknown_status_or_origin', source_url, 1
FROM classic_stable.auto_intelligence.market_download_manifest
WHERE status IS NULL OR status NOT IN ('downloaded', 'duplicate', 'not_found', 'failed')
   OR origin IS NULL OR origin NOT IN ('seed', 'discovery')
